            SVR_ROWS_PER_PARTICIPANT,
        "training_participants": len(training_people),
        "training_rows": len(balanced_train_indices),
        "validation_participants": len(validation_people),
        "validation_rows": len(validation_indices),
    })

cg_svr_pipeline = Pipeline([
    ("preprocessor", clone(cg_preprocessor)),
    (
        "model",
        SVR(
            kernel="rbf",
            epsilon=5.0,
            gamma="scale",
            cache_size=1024,
        ),
    ),
])

# refit=False prevents GridSearchCV from trying to train SVR
# on all 534,235 development rows.
cg_svr_search = GridSearchCV(
    estimator=cg_svr_pipeline,
    param_grid={
        "model__C": [1.0, 10.0],
    },
    scoring=regression_scores,
    refit=False,
    cv=cg_svr_cv_splits,
    n_jobs=1,
    verbose=2,
    error_score="raise",
)

print("Starting full participant-balanced CGMacros SVR tuning...")
print(
    "Maximum training rows per participant:",
    SVR_ROWS_PER_PARTICIPANT,
)

start = time.perf_counter()

cg_svr_search.fit(X_cg_dev, y_cg_dev)

tuning_minutes = (time.perf_counter() - start) / 60

cg_svr_search_results = pd.DataFrame(
    cg_svr_search.cv_results_
)

cg_svr_search_results.to_csv(
    results_folder / "cgmacros_svr_search.csv",
    index=False,
)

# MAE values are negative inside GridSearchCV.
# The largest value therefore represents the smallest MAE.
best_index = int(
    cg_svr_search_results["mean_test_MAE"].idxmax()
)

best_params = cg_svr_search_results.loc[
    best_index,
    "params",
]

best_c = float(best_params["model__C"])

cg_svr_rows = []

for fold_number in range(1, 6):
    fold_index = fold_number - 1

    cg_svr_rows.append({
        "dataset": "CGMacros",
        "model": "SVR",
        "training_protocol":
            "participant-balanced subset",
        "maximum_rows_per_participant":
            SVR_ROWS_PER_PARTICIPANT,
        "validation_fold": fold_number,
        "MAE": -cg_svr_search.cv_results_[
            f"split{fold_index}_test_MAE"
        ][best_index],
        "RMSE": -cg_svr_search.cv_results_[
            f"split{fold_index}_test_RMSE"
        ][best_index],
        "R2": cg_svr_search.cv_results_[
            f"split{fold_index}_test_R2"
        ][best_index],
    })

cg_svr_table = pd.DataFrame(cg_svr_rows)

cg_svr_table.to_csv(
    results_folder / "cgmacros_svr_cv.csv",
    index=False,
)

# Build the final saved SVR using a balanced sample from
# every development participant.
all_development_positions = np.arange(len(X_cg_dev))

cg_svr_final_indices = create_balanced_svr_sample(
    candidate_positions=all_development_positions,
    participant_series=groups_cg_dev,
    rows_per_participant=SVR_ROWS_PER_PARTICIPANT,
    random_seed=SVR_RANDOM_SEED,
)

assert set(
    groups_cg_dev.iloc[cg_svr_final_indices]
) == cg_dev_people

cg_svr_final_pipeline = Pipeline([
    ("preprocessor", clone(cg_preprocessor)),
    (
        "model",
        SVR(
            kernel="rbf",
            C=best_c,
            epsilon=5.0,
            gamma="scale",
            cache_size=1024,
        ),
    ),
])

print("\nBest settings:", {"model__C": best_c})
print("Training final balanced SVR pipeline...")

final_fit_start = time.perf_counter()

cg_svr_final_pipeline.fit(
    X_cg_dev.iloc[cg_svr_final_indices],
    y_cg_dev.iloc[cg_svr_final_indices],
)

final_fit_minutes = (
    time.perf_counter() - final_fit_start
) / 60

joblib.dump(
    cg_svr_final_pipeline,
    models_folder / "cgmacros_svr_pipeline.joblib",
)

cg_svr_protocol_rows.append({

    "maximum_rows_per_participant":
        SVR_ROWS_PER_PARTICIPANT,
    "training_participants":
        groups_cg_dev.iloc[cg_svr_final_indices].nunique(),
    "training_rows": len(cg_svr_final_indices),
    "validation_participants": 0,
    "validation_rows": 0,
})

cg_svr_protocol_table = pd.DataFrame(
    cg_svr_protocol_rows
)

cg_svr_protocol_table.to_csv(

print(cg_svr_table.round(4).to_string(index=False))
